\section{Model Formulation}

$\alpha$（促进系数）与 $\beta$（抑制系数）地位相同、结构对称，以拮抗关系分别作用于
logistic 模型的两个独立参数：

\begin{equation}\label{eq:ode}
\frac{dx}{dt} = \frac{\gamma}{1+\beta} \cdot x \cdot \left(1 - \frac{x}{K(1+\alpha)}\right)
\end{equation}

\begin{itemize}
  \item $\alpha \geq 0$（促进）：扩大有效饱和容量 $\;\rightarrow\; K_{\text{eff}} = K(1+\alpha)$
  \item $\beta \geq 0$（抑制）：降低有效增长率 $\;\rightarrow\; r = \gamma/(1+\beta)$
\end{itemize}

两者均以 $(1+c)^{\pm 1}$ 的函数形式（$c$ 为对应系数）体现对等地位；拮抗方向相反：
$\alpha$ 推高饱和平台，$\beta$ 拖慢趋近速度。当 $\alpha = \beta = 0$ 时退化为 Q1。

\section{Standard Logistic Form}

\begin{equation}\label{eq:logistic}
\frac{dx}{dt} = r \cdot x \cdot \left(1 - \frac{x}{K_{\text{eff}}}\right)
\end{equation}
\begin{equation}
r = \frac{\gamma}{1+\beta}, \qquad K_{\text{eff}} = K(1+\alpha)
\end{equation}

解析解：
\begin{equation}\label{eq:solution}
x(t) = \frac{K_{\text{eff}}}{1 + \left(\dfrac{K_{\text{eff}}}{x_0} - 1\right) e^{-r t}}
\end{equation}

\section{Parameter Identification}

式 (\ref{eq:solution}) 含两个独立拟合参数 $r$ 和 $K_{\text{eff}}$，拟合后：
\begin{equation}
\beta = \frac{\gamma}{r} - 1, \qquad \alpha = \frac{K_{\text{eff}}}{K} - 1
\end{equation}

$r \leq \gamma \Rightarrow \beta \geq 0$，$K_{\text{eff}} \geq K \Rightarrow \alpha \geq 0$，由拟合约束自然保证。

\end{itemize}

\section{Key Findings}

\begin{itemize}
  \item $\alpha = 0.0164 > 0$，$\beta = 1.5929 > 0$：两者均为正数，分别识别
  \item $\alpha$ 与 $\beta$ 地位对等：对称的 $(1+c)^{\pm 1}$ 结构
  \item 拮抗关系：$\alpha$ 扩展容量（$+1.64\%$），$\beta$ 抑制速率（降至 $38.6\%$）
  \item $K_{\text{eff}} = 8142$：促进将有效容量推至理论值 $K=8011$ 之上
  \item 结构稳定性：当 $\alpha, \beta \geq 0$ 时，$r > 0$ 恒成立，系统不存在分岔、不会崩溃
\end{itemize}
